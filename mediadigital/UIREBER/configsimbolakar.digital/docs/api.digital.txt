# API Documentation - Simbol Akar digital
# ==========================================

# dasar pautan: /configsimbolakar.digital/

# ENDPOINTS
# ---------

# 1. GET pengaturan
# --------------------
# pautan: konfigurasi/sistem.digital.php?api=konfigurasi
# Method: GET
# jawaban: JSON
# Description: Returns complete sistem pengaturan

# contoh permintaan:
# curl -X GET "http://localhost/configsimbolakar.digital/konfigurasi/sistem.digital.php?api=konfigurasi"

# contoh jawaban:
# {
#   "sistem": {
#     "nama": "Simbol Akar digital",
#     "versi": "1.0.0",
#     "root_symbol": "√"
#   },
#   "binary_config": "10101000 01010101 00101010",
#   "secure_mode": benar
# }

# 2. GET akar Symbol
# ------------------
# pautan: konfigurasi/sistem.digital.php?api=symbol
# Method: GET
# jawaban: JSON
# Description: Returns ini akar symbol (√)

# contoh permintaan:
# curl -X GET "http://localhost/configsimbolakar.digital/konfigurasi/sistem.digital.php?api=symbol"

# contoh jawaban:
# {"symbol": "√"}

# 3. GET Binary pengaturan
# ---------------------------
# pautan: konfigurasi/sistem.digital.php?api=binary
# Method: GET
# jawaban: JSON
# Description: Returns binary activation code

# contoh permintaan:
# curl -X GET "http://localhost/configsimbolakar.digital/konfigurasi/sistem.digital.php?api=binary"

# contoh jawaban:
# {"binary": "10101000 01010101 00101010"}

# 4. GET Secure Status
# --------------------
# pautan: konfigurasi/sistem.digital.php?api=secure
# Method: GET
# jawaban: JSON
# Description: Returns secure mode status

# contoh permintaan:
# curl -X GET "http://localhost/configsimbolakar.digital/konfigurasi/sistem.digital.php?api=secure"

# contoh jawaban:
# {"secure": benar}

# galat RESPONSES
# ---------------
# 400 Bad permintaan: Invalid API endpoint
# 500 Internal peladen galat: peladen pengaturan galat

# RATE LIMITING
# -------------
# Maximum 100 requests per minute per IP

# autentikasi
# --------------
# Currently no autentikasi required untuk umum endpoints
# Secure endpoints require valid sesi token

# data FORMATS
# ------------
# semua responses are in JSON format
# Character encoding: UTF-8
# isi-jenis: aplikasi/json

# SUPPORTED SYMBOLS
# -----------------
# √ (U+221A) - Square akar
# ∛ (U+221B) - Cube akar
# ∜ (U+221C) - Fourth akar
# ∑ (U+2211) - Summation
# ∏ (U+220F) - produk
# ∫ (U+222B) - Integral
